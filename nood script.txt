-- AIMBOT
local function ToggleAimbot(enabled)
    Features.Aimbot = enabled
    if enabled then
        -- Conectar aimbot
        AimConnection = RunService.RenderStepped:Connect(function()
            if not Features.Aimbot then return end
            
            local closest = nil
            local shortestDist = math.huge
            local mousePos = UserInputService:GetMouseLocation()
            
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("Head") then
                    local head = player.Character.Head
                    local pos, onScreen = Camera:WorldToViewportPoint(head.Position)
                    
                    if onScreen then
                        local dist = (Vector2.new(pos.X, pos.Y) - mousePos).Magnitude
                        if dist < Settings.AimFOV * 10 then
                            if dist < shortestDist then
                                shortestDist = dist
                                closest = head
                            end
                        end
                    end
                end
            end
            
            if closest then
                local pos = Camera:WorldToViewportPoint(closest.Position)
                -- Movimiento suave hacia el objetivo
                local current = UserInputService:GetMouseLocation()
                local target = Vector2.new(pos.X, pos.Y)
                local smooth = Vector2.new(
                    current.X + (target.X - current.X) * Settings.AimSmooth,
                    current.Y + (target.Y - current.Y) * Settings.AimSmooth
                )
                VirtualInputManager:SendMouseMoveEvent(smooth.X, smooth.Y, game)
            end
        end)
    else
        -- Desconectar aimbot
        if AimConnection then
            AimConnection:Disconnect()
            AimConnection = nil
        end
    end
end

-- ESP
local function CreateESP(player)
    if player == LocalPlayer then return end
    
    local box = Drawing.new("Square")
    local name = Drawing.new("Text")
    
    ESPObjects[player] = {Box = box, Name = name, Player = player}
end

local function ToggleESP(enabled)
    Features.ESP = enabled
    
    if enabled then
        -- Crear ESP para jugadores existentes
        for _, player in ipairs(Players:GetPlayers()) do
            CreateESP(player)
        end
        
        -- Conectar actualización
        RunService.RenderStepped:Connect(function()
            if not Features.ESP then
                for _, obj in pairs(ESPObjects) do
                    obj.Box.Visible = false
                    obj.Name.Visible = false
                end
                return
            end
            
            for _, obj in pairs(ESPObjects) do
                if obj.Player.Character and obj.Player.Character:FindFirstChild("HumanoidRootPart") then
                    local hrp = obj.Player.Character.HumanoidRootPart
                    local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                    
                    if onScreen then
                        local size = 1000 / pos.Z
                        obj.Box.Size = Vector2.new(size, size * 1.5)
                        obj.Box.Position = Vector2.new(pos.X - size/2, pos.Y - size*1.5/2)
                        obj.Box.Color = Color3.fromRGB(255, 50, 50)
                        obj.Box.Thickness = 1
                        obj.Box.Visible = true
                        
                        obj.Name.Text = obj.Player.Name
                        obj.Name.Position = Vector2.new(pos.X, pos.Y - size*1.5/2 - 15)
                        obj.Name.Color = Color3.fromRGB(255, 255, 255)
                        obj.Name.Size = 12
                        obj.Name.Center = true
                        obj.Name.Visible = true
                    else
                        obj.Box.Visible = false
                        obj.Name.Visible = false
                    end
                else
                    obj.Box.Visible = false
                    obj.Name.Visible = false
                end
            end
        end)
    else
        -- Ocultar todo
        for _, obj in pairs(ESPObjects) do
            obj.Box.Visible = false
            obj.Name.Visible = false
        end
    end
end

-- FLY
local function ToggleFly(enabled)
    Features.Fly = enabled
    
    if enabled then
        FlyConnection = RunService.RenderStepped:Connect(function()
            if not Features.Fly then return end
            
            local char = LocalPlayer.Character
            if not char or not char:FindFirstChild("HumanoidRootPart") then return end
            
            local hrp = char.HumanoidRootPart
            local cam = Workspace.CurrentCamera
            local velocity = Vector3.new(0, 0, 0)
            
            if UserInputService:IsKeyDown(Enum.KeyCode.W) then
                velocity = velocity + cam.CFrame.LookVector * Settings.FlySpeed
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.S) then
                velocity = velocity - cam.CFrame.LookVector * Settings.FlySpeed
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.A) then
                velocity = velocity - cam.CFrame.RightVector * Settings.FlySpeed
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.D) then
                velocity = velocity + cam.CFrame.RightVector * Settings.FlySpeed
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.Space) then
                velocity = velocity + Vector3.new(0, Settings.FlySpeed, 0)
            end
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then
                velocity = velocity - Vector3.new(0, Settings.FlySpeed, 0)
            end
            
            hrp.Velocity = velocity
            hrp.Anchored = velocity == Vector3.new(0, 0, 0)
        end)
    else
        if FlyConnection then
            FlyConnection:Disconnect()
            FlyConnection = nil
        end
        local char = LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.Anchored = false
        end
    end
end

-- SPEED
local function ToggleSpeed(enabled)
    Features.Speed = enabled
    SpeedControl.Visible = enabled
    
    local function updateSpeed()
        local char = LocalPlayer.Character
        if char and char:FindFirstChildOfClass("Humanoid") then
            char:FindFirstChildOfClass("Humanoid").WalkSpeed = enabled and Settings.SpeedValue or 16
        end
    end
    
    if enabled then
        updateSpeed()
        LocalPlayer.CharacterAdded:Connect(function(char)
            if Features.Speed then
                char:WaitForChild("Humanoid").WalkSpeed = Settings.SpeedValue
            end
        end)
    else
        updateSpeed()
    end
end

-- WALLHACK
local function ToggleWallhack(enabled)
    Features.Wallhack = enabled
    
    for _, part in ipairs(Workspace:GetDescendants()) do
        if part:IsA("BasePart") and not part:IsDescendantOf(LocalPlayer.Character) then
            if enabled then
                part.Transparency = 0.8
            else
                part.Transparency = 0
            end
        end
    end
end

-- CONEXIONES DE BOTONES (CLICK = ACTIVAR/DESACTIVAR)

local function UpdateButton(btn, feature, color)
    if feature then
        btn.BackgroundColor3 = color
        btn.Text = btn.Text:gsub("%[OFF%]", "[ON]")
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    else
        btn.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        btn.Text = btn.Text:gsub("%[ON%]", "[OFF]")
        btn.TextColor3 = Color3.fromRGB(200, 200, 200)
    end
end
